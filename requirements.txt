# GeoViewer - direct dependencies (Python 3.10, Windows x64).
# The exact, complete set used for the offline install is requirements-offline.txt.

# --- runtime ----------------------------------------------------------------
numpy               # arrays for every reader and the block-face extraction
pandas              # CSV / drillhole tables
pyarrow             # fast multi-threaded CSV reading for large block models
vtk                 # OpenGL rendering engine
pyvista             # pythonic VTK meshes / plotter
pyvistaqt           # VTK render window inside a Qt widget
PySide6-Essentials  # Qt 6 GUI (QtCore, QtGui, QtWidgets) - LGPL
QtAwesome           # Material Design icons for the toolbar
ezdxf               # DXF support library (old R10 DXF is read by geoviewer/io/dxf_reader.py)
access-parser       # reads Surpac drillhole .accdb/.mdb without an ODBC driver
PyMySQL[rsa]        # MySQL server as the project database (optional setting; rsa = MySQL 8 sign-in)
openpyxl            # Excel (.xlsx) export of the drillhole data

# --- development ------------------------------------------------------------
pytest

# --- build / release ----------------------------------------------------------
pyinstaller
cryptography>=42         # license key verification (Ed25519)
